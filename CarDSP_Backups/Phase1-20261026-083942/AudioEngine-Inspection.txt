 209: bool AudioPlaybackEngine::playPcm(
 210:     const QString &path,
 211:     int deviceIndex)
 212: {
 213:     stop();
 214:     setError(QString());
 215: 
 216:     if (deviceIndex < 0 || deviceIndex >= m_devices.size()) {
 217:         setError("Invalid output device index.");
 218:         return false;
 219:     }
 220: 
 221:     auto file = std::make_unique<QFile>(path);
 222: 
 223:     if (!file->open(QIODevice::ReadOnly)) {
 224:         setError("Unable to open PCM audio file.");
 225:         return false;
 226:     }
 227: 
 228:     if (file->size() % 8 != 0) {
 229:         setError(
 230:             "PCM file must contain complete stereo Float32 frames."
 231:         );
 232:         return false;
 233:     }
 234: 
 235:     QAudioFormat format;
 236:     format.setSampleRate(48000);
 237:     format.setChannelCount(2);
 238:     format.setSampleFormat(QAudioFormat::Float);
 239: 
 240:     const QAudioDevice device = m_devices[deviceIndex];
 241: 
 242:     if (!device.isFormatSupported(format)) {
 243:         setError(
 244:             "Output device does not support 48 kHz stereo Float32."
 245:         );
 246:         return false;
 247:     }
 248: 
 249:     m_file = std::move(file);
 250:     m_processor.clear();
 251: 
 252:     constexpr float sampleRate = 48000.0f;
 253: 
 254:     for (const QVariant &entry : m_eqBands) {
 255:         const QVariantMap band = entry.toMap();
 256: 
 257:         if (!band.value("enabled", true).toBool())
 258:             continue;
 259: 
 260:         const float frequency =
 261:             band.value("frequency", 1000.0).toFloat();
 262: 
 263:         const float gain =
 264:             band.value("gain", 0.0).toFloat();
 265: 
 266:         const float q =
 267:             band.value("q", 1.41).toFloat();
 268: 
 269:         const int type =
 270:             band.value("type", 0).toInt();
 271: 
 272:         cardsp::BiquadCoefficients coefficients;
 273: 
 274:         switch (type) {
 275:         case 0:
 276:             coefficients = cardsp::makePeakingEQ(
 277:                 sampleRate, frequency, q, gain);
 278:             break;
 279:         case 1:
 280:             coefficients = cardsp::makeLowShelf(
 281:                 sampleRate, frequency, gain);
 282:             break;
 283:         case 2:
 284:             coefficients = cardsp::makeHighShelf(
 285:                 sampleRate, frequency, gain);
 286:             break;
 287:         case 3:
 288:             coefficients = cardsp::makeLowPass(
 289:                 sampleRate, frequency, q);
 290:             break;
 291:         case 4:
 292:             coefficients = cardsp::makeHighPass(
 293:                 sampleRate, frequency, q);
 294:             break;
 295:         case 5:
 296:             coefficients = cardsp::makeNotch(
 297:                 sampleRate, frequency, q);
 298:             break;
 299:         default:
 300:             setError("Unsupported EQ filter type.");
 301:             stop();
 302:             return false;
 303:         }
 304: 
 305:         if (!m_processor.add(coefficients)) {
 306:             setError("Unable to add EQ filter.");
 307:             stop();
 308:             return false;
 309:         }
 310:     }
 311: 
 312:     m_processor.reset();
 313: 
 314:     m_sink = std::make_unique<QAudioSink>(
 315:         device,
 316:         format,
 317:         this
 318:     );
 319: 
 320:     m_sink->setVolume(static_cast<float>(m_volume));
 321: 
 322:     m_stream = std::make_unique<PcmStreamDevice>(
 323:         m_file.get(),
 324:         &m_processor
 325:     );
 326: 
 327:     m_stream->setBypass(m_bypass);
 328: 
 329:     if (!m_stream->open(QIODevice::ReadOnly)) {
 330:         setError("Unable to open PCM stream.");
 331:         stop();
 332:         return false;
 333:     }
 334: 
 335:     connect(
 336:         m_sink.get(),
 337:         &QAudioSink::stateChanged,
 338:         this,
 339:         [this](QAudio::State state) {
 340:             qInfo() << "CarDSP sink state:"
 341:                     << static_cast<int>(state)
 342:                     << "error="
 343:                     << (m_sink ? static_cast<int>(m_sink->error()) : -1)
 344:                     << "filePosition="
 345:                     << (m_file ? m_file->pos() : -1)
 346:                     << "fileSize="
 347:                     << (m_file ? m_file->size() : -1);
 348: 
 349:             if (!m_sink)
 350:                 return;
 351: 
 352:             if (state == QAudio::IdleState) {
 353:                 if (m_file && m_file->atEnd()) {
 354:                     qInfo() << "CarDSP: End of PCM file reached.";
 355:                     stop();
 356:                 } else {
 357:                     qWarning() << "CarDSP: Sink idle before EOF.";
 358:                 }
 359:             } else if (state == QAudio::StoppedState &&
 360:                        m_sink->error() != QAudio::NoError) {
 361:                 setError("Audio output stopped with an error.");
 362:                 stop();
 363:             }
 364:         },
 365:         Qt::QueuedConnection
 366:     );
 367: 
 368:     qInfo() << "CarDSP output device:" << device.description();
 369:     qInfo() << "CarDSP output format:"
 370:             << format.sampleRate()
 371:             << format.channelCount()
 372:             << format.sampleFormat();
 373: 
 374:     qInfo() << "CarDSP TEST: bypassing custom PCM stream";
 375:     m_sink->start(m_file.get());
 376: 
 377:     qInfo() << "CarDSP sink after start:"
 378:             << "state=" << static_cast<int>(m_sink->state())
 379:             << "error=" << static_cast<int>(m_sink->error())
 380:             << "processedUSecs=" << m_sink->processedUSecs();
 381: 
 382:     if (m_sink->error() != QAudio::NoError) {
 383:         setError("Audio output failed to start.");
 384:         stop();
 385:         return false;
 386:     }
 387: 
 388:     m_playing = true;
 389:     emit playingChanged();
 390: 
 391:     return true;
