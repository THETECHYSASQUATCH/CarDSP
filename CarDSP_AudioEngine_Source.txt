class AudioPlaybackEngine : public QObject
{
    Q_OBJECT

    Q_PROPERTY(bool playing READ playing NOTIFY playingChanged)
    Q_PROPERTY(bool bypass READ bypass WRITE setBypass NOTIFY bypassChanged)
    Q_PROPERTY(double volume READ volume WRITE setVolume NOTIFY volumeChanged)
    Q_PROPERTY(QString lastError READ lastError NOTIFY lastErrorChanged)
    Q_PROPERTY(QVariantList outputDevices READ outputDevices NOTIFY outputDevicesChanged)

public:
    explicit AudioPlaybackEngine(QObject *parent = nullptr);
    ~AudioPlaybackEngine() override;

    bool playing() const { return m_playing; }
    bool bypass() const { return m_bypass; }
    double volume() const { return m_volume; }
    QString lastError() const { return m_lastError; }

    QVariantList outputDevices() const;
    Q_INVOKABLE QString localFilePath(const QUrl &url) const { return url.toLocalFile(); }

    Q_INVOKABLE bool playPcm(const QString &path, int deviceIndex = 0);
    Q_INVOKABLE bool configureEq(const QVariantList &bands);
    Q_INVOKABLE void stop();
    Q_INVOKABLE void refreshOutputDevices();

    void setBypass(bool bypass);
    void setVolume(double volume);

signals:
    void playingChanged();
    void bypassChanged();
    void volumeChanged();
    void lastErrorChanged();
    void outputDevicesChanged();

private:
    void setError(const QString &message);

    QMediaDevices m_mediaDevices;
    QList<QAudioDevice> m_devices;

    std::unique_ptr<QAudioSink> m_sink;
    std::unique_ptr<PcmStreamDevice> m_stream;
    std::unique_ptr<QFile> m_file;

    cardsp::StereoProcessor<> m_processor;
    QVariantList m_eqBands;

    bool m_playing = false;
    bool m_bypass = false;
    double m_volume = 0.25;
    QString m_lastError;
};



===== UI\Desktop\Audio\AudioPlaybackEngine.cpp =====
#include "AudioPlaybackEngine.hpp"

#include <QAudio>
#include <QIODevice>
#include <QVariantMap>

#include <algorithm>
#include <array>
#include <cstring>

class PcmStreamDevice final : public QIODevice
{
public:
    PcmStreamDevice(
        QFile *file,
        cardsp::StereoProcessor<> *processor,
        QObject *parent = nullptr)
        : QIODevice(parent),
          m_file(file),
          m_processor(processor)
    {
    }

    void setBypass(bool bypass) noexcept
    {
        m_bypass = bypass;
    }

    bool atEnd() const override
    {
        return !m_file || m_file->atEnd();
    }

    bool isSequential() const override
    {
        return true;
    }

protected:
    qint64 readData(char *data, qint64 maxSize) override
    {
        if (!m_file || !data || maxSize <= 0)
            return 0;

        constexpr qint64 frameBytes = 8;
        constexpr qint64 bufferBytes = 8192;

        const qint64 requested =
            std::min((maxSize / frameBytes) * frameBytes,
                     bufferBytes);

        if (requested <= 0)
            return 0;

        const qint64 bytesRead =
            m_file->read(data, requested);

        if (bytesRead <= 0)
            return bytesRead;

        if (!m_bypass && m_processor) {
            const std::size_t sampleCount =
                static_cast<std::size_t>(
                    bytesRead / sizeof(float)
                );

            const std::size_t frameCount =
                sampleCount / 2;

            std::memcpy(
                m_samples.data(),
                data,
                static_cast<std::size_t>(bytesRead)
            );

            m_processor->processInterleaved(
                m_samples.data(),
                frameCount,
                2
            );

            std::memcpy(
                data,
                m_samples.data(),
                static_cast<std::size_t>(bytesRead)
            );
        }

        return bytesRead;
    }

    qint64 writeData(const char *, qint64) override
    {
        return -1;
    }

private:
    QFile *m_file = nullptr;
    cardsp::StereoProcessor<> *m_processor = nullptr;

    bool m_bypass = false;

    // 8192 bytes / 4 bytes per Float32 sample.
    std::array<float, 2048> m_samples{};
};

AudioPlaybackEngine::AudioPlaybackEngine(QObject *parent)
    : QObject(parent)
{
    refreshOutputDevices();

    connect(
        &m_mediaDevices,
        &QMediaDevices::audioOutputsChanged,
        this,
        &AudioPlaybackEngine::refreshOutputDevices
    );
}

AudioPlaybackEngine::~AudioPlaybackEngine()
{
    stop();
}

QVariantList AudioPlaybackEngine::outputDevices() const
{
    QVariantList result;

    for (int i = 0; i < m_devices.size(); ++i) {
        QVariantMap item;
        item["index"] = i;
        item["name"] = m_devices[i].description();
        result.append(item);
    }

    return result;
}

void AudioPlaybackEngine::refreshOutputDevices()
{
    m_devices = QMediaDevices::audioOutputs();
    emit outputDevicesChanged();
}

void AudioPlaybackEngine::setError(const QString &message)
{
    if (m_lastError == message)
        return;

    m_lastError = message;
    emit lastErrorChanged();
}

void AudioPlaybackEngine::setBypass(bool bypass)
{
    if (m_bypass == bypass)
        return;

    m_bypass = bypass;

    if (m_stream)
        m_stream->setBypass(bypass);

    emit bypassChanged();
}

void AudioPlaybackEngine::setVolume(double volume)
{
    volume = std::clamp(volume, 0.0, 1.0);

    if (m_volume == volume)
        return;

    m_volume = volume;

    if (m_sink)
        m_sink->setVolume(static_cast<float>(volume));

    emit volumeChanged();
}

bool AudioPlaybackEngine::configureEq(const QVariantList &bands)
{
    if (m_playing) {
        setError("Stop playback before configuring EQ.");
        return false;
    }

    if (bands.size() > 40) {
        setError("Too many EQ filters.");
        return false;
    }

    m_eqBands = bands;
    setError(QString());
    return true;
}
bool AudioPlaybackEngine::playPcm(
    const QString &path,
    int deviceIndex)
{
    stop();
    setError(QString());

    if (deviceIndex < 0 || deviceIndex >= m_devices.size()) {
        setError("Invalid output device index.");
        return false;
    }

    auto file = std::make_unique<QFile>(path);

    if (!file->open(QIODevice::ReadOnly)) {
        setError("Unable to open PCM audio file.");
        return false;
    }

    if (file->size() % 8 != 0) {
        setError(
            "PCM file must contain complete stereo Float32 frames."
        );
        return false;
    }

    QAudioFormat format;
    format.setSampleRate(48000);
    format.setChannelCount(2);
    format.setSampleFormat(QAudioFormat::Float);

    const QAudioDevice device = m_devices[deviceIndex];

    if (!device.isFormatSupported(format)) {
        setError(
            "Output device does not support 48 kHz stereo Float32."
        );
        return false;
    }

    m_file = std::move(file);
    m_processor.clear();

    constexpr float sampleRate = 48000.0f;

    for (const QVariant &entry : m_eqBands) {
        const QVariantMap band = entry.toMap();

        if (!band.value("enabled", true).toBool())
            continue;

        const float frequency =
            band.value("frequency", 1000.0).toFloat();

        const float gain =
            band.value("gain", 0.0).toFloat();

        const float q =
            band.value("q", 1.41).toFloat();

        const int type =
            band.value("type", 0).toInt();

        cardsp::BiquadCoefficients coefficients;

        switch (type) {
        case 1:
            coefficients = cardsp::makeLowPass(
                sampleRate, frequency, q);
            break;

        case 2:
            coefficients = cardsp::makeHighPass(
                sampleRate, frequency, q);
            break;

        default:
            coefficients = cardsp::makePeakingEQ(
                sampleRate, frequency, q, gain);
            break;
        }

        if (!m_processor.add(coefficients)) {
            setError("Unable to add EQ filter.");
            stop();
            return false;
        }
    }

    m_processor.reset();

    m_sink = std::make_unique<QAudioSink>(
        device,
        format,
        this
    );

    m_sink->setVolume(static_cast<float>(m_volume));

    m_stream = std::make_unique<PcmStreamDevice>(
        m_file.get(),
        &m_processor
    );

    m_stream->setBypass(m_bypass);

    if (!m_stream->open(QIODevice::ReadOnly)) {
        setError("Unable to open PCM stream.");
        stop();
        return false;
    }

    connect(
        m_sink.get(),
        &QAudioSink::stateChanged,
        this,
        [this](QAudio::State state) {
            if (state == QAudio::IdleState) {
                stop();
            } else if (state == QAudio::StoppedState &&
                       m_sink &&
                       m_sink->error() != QAudio::NoError) {
                setError("Audio output stopped with an error.");
                stop();
            }
        },
        Qt::QueuedConnection
    );

    m_sink->start(m_stream.get());

    if (m_sink->error() != QAudio::NoError) {
        setError("Audio output failed to start.");
        stop();
        return false;
    }

    m_playing = true;
    emit playingChanged();

    return true;
}

void AudioPlaybackEngine::stop()
{
    if (m_sink) {
        m_sink->stop();
        m_sink.reset();
    }

    if (m_stream) {
        m_stream->close();
        m_stream.reset();
    }

    m_file.reset();

    if (m_playing) {
        m_playing = false;
        emit playingChanged();
    }
}
